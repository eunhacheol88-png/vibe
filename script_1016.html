핵심 정리: Script Properties와 Drive·Docs·PDF·Calendar·Gmail 연결을(를) 현재 운영관리 시스템의 데이터 흐름과 연결해서 이해합니다.

THEORY 12. 웹앱 배포와 컨테이너 바인딩의 차이

핵심 개념

핵심 정리: 웹앱 배포와 컨테이너 바인딩의 차이을(를) 현재 운영관리 시스템의 데이터 흐름과 연결해서 이해합니다.

function onOpen() {
  const ui = SpreadsheetApp.getUi();

  ui.createMenu('📋 복지 프로그램 관리')
    .addItem('안내 메시지 보기', 'showWelcomeAlert')
    .addItem('담당자 입력하기', 'showManagerPrompt')
    .addItem('작업 완료 알림', 'showToastMessage')
    .addSeparator()
    .addItem('업무 안내창 열기', 'showModalGuide')
    .addItem('업무 패널 열기', 'showSidebar')
    .addToUi();
}

function showWelcomeAlert() {
  const ui = SpreadsheetApp.getUi();

  const result = ui.alert(
    '업무 확인',
    '프로그램 운영 데이터를 정리하시겠습니까?',
    ui.ButtonSet.YES_NO
  );

  if (result === ui.Button.YES) {
    SpreadsheetApp.getActiveSpreadsheet().toast(
      '정리 작업을 시작합니다.',
      '업무 자동화',
      3
    );
  } else {
    SpreadsheetApp.getActiveSpreadsheet().toast(
      '작업을 취소했습니다.',
      '업무 자동화',
      3
    );
  }
}

function showManagerPrompt() {
  const ui = SpreadsheetApp.getUi();

  const response = ui.prompt(
    '담당자 입력',
    '담당자 이름을 입력해주세요.',
    ui.ButtonSet.OK_CANCEL
  );

  if (response.getSelectedButton() === ui.Button.OK) {
    const managerName = response.getResponseText();

    SpreadsheetApp.getActiveSpreadsheet().toast(
      managerName + ' 담당자로 입력되었습니다.',
      '담당자 설정',
      3
    );
  }
}

function showToastMessage() {
  SpreadsheetApp.getActiveSpreadsheet().toast(
    '프로그램 업무 처리가 완료되었습니다.',
    '완료',
    5
  );
}

function showModalGuide() {
  const html = HtmlService
    .createHtmlOutputFromFile('Guide')
    .setWidth(420)
    .setHeight(350);

  SpreadsheetApp.getUi().showModalDialog(
    html,
    '업무 안내'
  );
}

function showSidebar() {
  const html = HtmlService
    .createHtmlOutputFromFile('Sidebar')
    .setTitle('복지 프로그램 업무 패널');

  SpreadsheetApp.getUi().showSidebar(html);
}

function checkProgramStatus() {

  console.log(
    '===== 프로그램 운영 현황 확인 시작 ====='
  );

  try {

    const SPREADSHEET_ID =
      PropertiesService
        .getScriptProperties()
        .getProperty(
          'SPREADSHEET_ID'
        );

    console.log(
      'SPREADSHEET_ID 설정값을 확인했습니다.'
    );

    if (!SPREADSHEET_ID) {

      throw new Error(
        'SPREADSHEET_ID가 설정되어 있지 않습니다.'
      );
    }

    const ss =
      SpreadsheetApp.openById(
        SPREADSHEET_ID
      );

    console.log(
      '스프레드시트를 열었습니다.'
    );

    const sheet =
      ss.getSheetByName(
        '프로그램운영관리'
      );

    if (!sheet) {

      throw new Error(
        '프로그램운영관리 시트를 찾을 수 없습니다.'
      );
    }

    console.log(
      '프로그램운영관리 시트를 찾았습니다.'
    );

    const lastRow =
      sheet.getLastRow();

    console.log(
      '현재 마지막 행: ' +
      lastRow
    );

    if (lastRow < 2) {

      console.log(
        '확인할 데이터가 없습니다.'
      );

      return {
        total: 0,
        waiting: 0,
        progressing: 0,
        completed: 0,
        paused: 0
      };
    }

    const data =
      sheet
        .getRange(
          2,
          1,
          lastRow - 1,
          10
        )
        .getValues();

    console.log(
      '전체 범위 데이터를 가져왔습니다.'
    );

    const validData =
      data.filter(
        function(rowData) {
          return rowData[0] !== '';
        }
      );

    console.log(
      '실제 집계 대상: ' +
      validData.length +
      '건'
    );

    let waiting = 0;
    let progressing = 0;
    let completed = 0;
    let paused = 0;

    validData.forEach(
      function(rowData) {

        const status =
          rowData[5];

        if (
          status === '대기'
        ) {
          waiting++;
        }

        else if (
          status === '진행중'
        ) {
          progressing++;
        }

        else if (
          status === '완료'
        ) {
          completed++;
        }

        else if (
          status === '보류'
        ) {
          paused++;
        }
      }
    );

    const result = {
      total: validData.length,
      waiting: waiting,
      progressing: progressing,
      completed: completed,
      paused: paused
    };

    console.log(
      '전체: ' +
      result.total +
      '명'
    );

    console.log(
      '대기: ' +
      result.waiting +
      '명'
    );

    console.log(
      '진행중: ' +
      result.progressing +
      '명'
    );

    console.log(
      '완료: ' +
      result.completed +
      '명'
    );

    console.log(
      '보류: ' +
      result.paused +
      '명'
    );

    console.log(
      '===== 프로그램 운영 현황 확인 완료 ====='
    );

    return result;

  } catch (error) {

    console.error(
      '운영 현황 확인 오류: ' +
      error.message
    );

    throw error;
  }
}

function formatProgramSheet() {

  console.log(
    '===== 업무 시트 서식 정리 시작 ====='
  );

  try {

    const ss =
      SpreadsheetApp.getActiveSpreadsheet();

    const sheet =
      ss.getSheetByName(
        '프로그램운영관리'
      );

    if (!sheet) {

      throw new Error(
        '프로그램운영관리 시트를 찾을 수 없습니다.'
      );
    }

    console.log(
      '프로그램운영관리 시트를 찾았습니다.'
    );

    // 1. K열 제목
    sheet
      .getRange('K1')
      .setValue('후속확인');

    console.log(
      'K열 후속확인 제목을 설정했습니다.'
    );

    // 2. 제목 행 서식
    const headerRange =
      sheet.getRange(
        1,
        1,
        1,
        11
      );

    headerRange
      .setFontWeight('bold')
      .setFontColor('#ffffff')
      .setBackground('#355E3B')
      .setHorizontalAlignment('center');

    console.log(
      '제목 행 서식을 적용했습니다.'
    );

    // 3. 첫 행 고정
    sheet.setFrozenRows(1);

    console.log(
      '첫 번째 행을 고정했습니다.'
    );

    // 4. 날짜 형식
    sheet
      .getRange('D2:D')
      .setNumberFormat('yyyy-MM-dd');

    sheet
      .getRange('G2:G')
      .setNumberFormat('yyyy-MM-dd');

    sheet
      .getRange('H2:H')
      .setNumberFormat('yyyy-MM-dd');

    console.log(
      '날짜 형식을 yyyy-MM-dd로 정리했습니다.'
    );

    // 5. F열 상태 드롭다운
    const statusRule =
      SpreadsheetApp
        .newDataValidation()
        .requireValueInList(
          [
            '대기',
            '진행중',
            '완료',
            '보류'
          ],
          true
        )
        .setAllowInvalid(false)
        .build();

    sheet
      .getRange('F2:F1000')
      .setDataValidation(
        statusRule
      );

    console.log(
      '상태 드롭다운을 설정했습니다.'
    );

    // 6. K열 체크박스 규칙
    const checkboxRange =
      sheet.getRange(
        'K2:K1000'
      );

    const checkboxRule =
      SpreadsheetApp
        .newDataValidation()
        .requireCheckbox()
        .build();

    const checkboxRules =
      Array.from(
        {
          length:
            checkboxRange.getNumRows()
        },
        function() {
          return [
            checkboxRule
          ];
        }
      );

    checkboxRange
      .setDataValidations(
        checkboxRules
      );

    console.log(
      '후속확인 체크박스를 설정했습니다.'
    );

    // 7. 기존 필터 제거
    const existingFilter =
      sheet.getFilter();

    if (
      existingFilter
    ) {

      existingFilter.remove();

      console.log(
        '기존 필터를 제거했습니다.'
      );
    }

    // 8. A~K 필터 적용
    const lastRow =
      Math.max(
        sheet.getLastRow(),
        2
      );

    sheet
      .getRange(
        1,
        1,
        lastRow,
        11
      )
      .createFilter();

    console.log(
      'A~K 범위에 필터를 설정했습니다.'
    );

    // 9. 열 너비
    sheet.setColumnWidth(1, 90);
    sheet.setColumnWidth(2, 110);
    sheet.setColumnWidth(3, 170);
    sheet.setColumnWidth(4, 100);
    sheet.setColumnWidth(5, 100);
    sheet.setColumnWidth(6, 90);
    sheet.setColumnWidth(7, 100);
    sheet.setColumnWidth(8, 100);
    sheet.setColumnWidth(9, 200);
    sheet.setColumnWidth(10, 240);
    sheet.setColumnWidth(11, 90);

    console.log(
      '열 너비를 정리했습니다.'
    );

    // 10. 긴 내용 줄바꿈
    sheet
      .getRange('I:J')
      .setWrap(true);

    console.log(
      '비고와 파일링크에 줄바꿈을 적용했습니다.'
    );

    SpreadsheetApp.flush();

    ss.toast(
      '업무 시트 서식이 정리되었습니다.',
      '서식 정리 완료',
      4
    );

    console.log(
      '===== 업무 시트 서식 정리 완료 ====='
    );

  } catch (error) {

    console.error(
      '업무 시트 서식 정리 오류: ' +
      error.message
    );

    throw error;
  }
}

function createProgramSummaryReport() {

  console.log('===== 프로그램 운영 요약 보고서 생성 시작 =====');

  try {

    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // STEP 5에서 만든 기존 집계 함수를 다시 사용합니다.
    // 이 함수는 전체/대기/진행중/완료/보류 인원수를 계산해 돌려줍니다.
    const result = checkProgramStatus();

    console.log(
      '집계 결과: ' +
      JSON.stringify(result)
    );

    let reportSheet =
      ss.getSheetByName('프로그램요약보고서');

    console.log(
      '프로그램요약보고서 시트 존재 여부: ' +
      Boolean(reportSheet)
    );

    if (!reportSheet) {

      reportSheet =
        ss.insertSheet('프로그램요약보고서');

      console.log(
        '프로그램요약보고서 시트를 새로 만들었습니다.'
      );

    } else {

      // 기존 보고서가 있다면 병합된 셀을 먼저 풀어
      // 다시 만들 때 구조가 꼬이지 않도록 합니다.

        .getDataRange()
        .breakApart();

        '기존 보고서의 병합을 해제했습니다.'
      );

      );
    }

    const titleRange =
      reportSheet.getRange('A1:D2');

    titleRange.merge();

    titleRange
      .setValue('프로그램 운영 현황 보고서')
      .setFontSize(16)
      .setFontWeight('bold')
      .setHorizontalAlignment('center')
      .setVerticalAlignment('middle');

      'A1:D2 제목 영역 병합 및 제목 입력 완료'

    );

  } catch (error) {

    console.error(
      '보고서 생성 오류: ' +
      error.message
    );

  }

왜 수정하나요?
기존 코드에서 getDataRange().breakApart()를 사용하면 이미 병합된 제목 영역 일부만 선택되는 상황에서 “해당 범위의 모든 셀을 선택해야 합니다” 오류가 날 수 있습니다. 이제 A1:D2를 정확히 지정해 병합을 해제하도록 수정합니다.

function createProgramSummaryReport() {

  console.log('===== 프로그램 운영 요약 보고서 생성 시작 =====');

  try {

    const ss =
      SpreadsheetApp.getActiveSpreadsheet();

    console.log(
      '현재 스프레드시트: ' +

        'N열에 메일 발송 시간을 기록했습니다.'
      );

    } else {

      console.log(
        '이미 메일 발송 기록이 있으므로 이메일 발송을 건너뜁니다.'
      );
    }

    SpreadsheetApp.flush();

    ss.toast(
      '상담 일정과 이메일 처리를 완료했습니다.',
      '상담 업무 자동화',
      5
    );

    console.log(
      '===== 상담 일정 등록 및 이메일 발송 정상 종료 ====='
    );

  } catch (error) {

    console.error(
      '상담 일정/이메일 처리 오류: ' +
      error.message
    );

    SpreadsheetApp
      .getUi()
      .alert(
        '상담 업무 처리 오류\n\n' +
        error.message
      );

    throw error;
  }
}
